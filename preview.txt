// 1. Construct request body
let body = {};
const fields = ["firstname", "lastname", "totalprice", "depositpaid", "additionalneeds"];
fields.forEach(field => {
    const val = pm.iterationData.get(field);
    if (val !== "OMIT" && val !== undefined) {
        if (field === "totalprice") {
            body[field] = parseInt(val, 10);
        } else if (field === "depositpaid") {
            body[field] = val === "true" || val === true;
        } else {
            body[field] = val;
        }
    }
});
const checkin = pm.iterationData.get("checkin");
const checkout = pm.iterationData.get("checkout");
if ((checkin !== "OMIT" && checkin !== undefined) || (checkout !== "OMIT" && checkout !== undefined)) {
    body.bookingdates = {};
    if (checkin !== "OMIT" && checkin !== undefined) body.bookingdates.checkin = checkin;
    if (checkout !== "OMIT" && checkout !== undefined) body.bookingdates.checkout = checkout;
}

// 2. Set dynamic variables
pm.variables.set("dynamicBody", JSON.stringify(body));

    const createdId = pm.variables.get("dynamic_created_id");

    // 2. Verify status code

        pm.response.to.have.status(200);
    });
    let responseJson;

    // 3. Parse JSON response
    try {
        responseJson = pm.response.json();
    } catch (e) {
        pm.expect.fail("Invalid JSON");
        return;
    }
    pm.test(`[${scenario}] Response is an array`, function() {
        pm.expect(responseJson).to.be.an("array");
    });

    // 4. Run scenario assertions
    if (shouldFindMatch) {
        pm.test(`[${scenario}] Match found in results (contains created ID: ${createdId})`, function() {
            const found = responseJson.find(b => b.bookingid === createdId);
            pm.expect(found).to.not.be.undefined;
        });
    } else {
        pm.test(`[${scenario}] No matches found for invalid filters`, function() {
            const found = responseJson.find(b => b.bookingid === createdId);
            pm.expect(found).to.be.undefined;
        });
    }
} finally {
    // 5. Cleanup variables
    pm.variables.unset("dynamic_created_id");
}
